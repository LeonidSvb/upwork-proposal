multi_signal_indicator.pine 
// Multi-Signal Indicator v1.0
// EMA Crossover + RSI Filter + Volume Confirmation
//@version=5
indicator("Multi-Signal Indicator", overlay=true, max_labels_count=500)

// --- Inputs ---
ema_fast_len  = input.int(20,   "EMA Fast",    minval=1)
ema_slow_len  = input.int(50,   "EMA Slow",    minval=1)
rsi_len       = input.int(14,   "RSI Length",  minval=2)
rsi_ob        = input.float(70.0, "RSI Overbought")
rsi_os        = input.float(30.0, "RSI Oversold")
vol_mult      = input.float(1.5,  "Volume Multiplier", minval=0.5)
show_table    = input.bool(true,  "Show Stats Table")

// --- Calculations ---
ema_fast = ta.ema(close, ema_fast_len)
ema_slow = ta.ema(close, ema_slow_len)
rsi_val  = ta.rsi(close, rsi_len)
vol_avg  = ta.sma(volume, 20)

// --- Signal Conditions ---
cross_up   = ta.crossover(ema_fast, ema_slow)
cross_dn   = ta.crossunder(ema_fast, ema_slow)
vol_ok     = volume > vol_avg * vol_mult
rsi_buy    = rsi_val < (rsi_ob - 15)
rsi_sell   = rsi_val > (rsi_os + 15)

buy_signal  = cross_up and rsi_buy  and vol_ok
sell_signal = cross_dn and rsi_sell and vol_ok

// --- Plots ---
plot(ema_fast, "EMA Fast", color=color.new(color.teal,   20), linewidth=2)
plot(ema_slow, "EMA Slow", color=color.new(color.orange, 20), linewidth=2)

plotshape(buy_signal,  style=shape.triangleup,
          location=location.belowbar, color=color.lime,
          size=size.small, title="BUY",  text="BUY",  textcolor=color.lime)

plotshape(sell_signal, style=shape.triangledown,
          location=location.abovebar, color=color.red,
          size=size.small, title="SELL", text="SELL", textcolor=color.red)

// --- Background Highlight ---
bgcolor(buy_signal  ? color.new(color.lime, 92) : na)
bgcolor(sell_signal ? color.new(color.red,  92) : na)

// --- Alerts ---
alertcondition(buy_signal,  title="BUY Signal",  message="Multi-Signal: BUY — {{ticker}} {{interval}}")
alertcondition(sell_signal, title="SELL Signal", message="Multi-Signal: SELL — {{ticker}} {{interval}}")

// --- Stats Table ---
if show_table and barstate.islast
    t = table.new(position.top_right, 2, 5,
                  bgcolor=color.new(color.black, 70),
                  border_width=1, border_color=color.gray)
    table.cell(t, 0, 0, "Indicator",        text_color=color.gray,  text_size=size.small)
    table.cell(t, 1, 0, "Multi-Signal v1", text_color=color.white, text_size=size.small)
    table.cell(t, 0, 1, "EMA Fast/Slow",    text_color=color.gray,  text_size=size.small)
    table.cell(t, 1, 1, str.tostring(ema_fast_len) + " / " + str.tostring(ema_slow_len),
                        text_color=color.teal,  text_size=size.small)
    table.cell(t, 0, 2, "RSI",              text_color=color.gray,  text_size=size.small)
    table.cell(t, 1, 2, str.tostring(math.round(rsi_val, 1)),
                        text_color=color.yellow, text_size=size.small)
    table.cell(t, 0, 3, "Vol Filter",       text_color=color.gray,  text_size=size.small)
    table.cell(t, 1, 3, vol_ok ? "ACTIVE" : "—",
                        text_color=vol_ok ? color.lime : color.gray, text_size=size.small)
    table.cell(t, 0, 4, "Signal",           text_color=color.gray,  text_size=size.small)
    table.cell(t, 1, 4, buy_signal ? "BUY" : sell_signal ? "SELL" : "—",
                        text_color=buy_signal ? color.lime : sell_signal ? color.red : color.gray,
                        text_size=size.small)
